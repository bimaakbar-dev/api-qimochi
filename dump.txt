  push:
    branches: [main]
  repository_dispatch:
    types: [data-updated]

    runs-on: ubuntu-latest
    timeout-minutes: 10

      - name: Checkout api-qimochi
        uses: actions/checkout@v7

      - name: Checkout qimochi-data

          token: ${{ secrets.QIMOCHI_DATA_PAT }}

      - name: Inject data
        run: |
          rm -rf src/content/anime
          mkdir -p src/content/anime
          if [ -d _data/src/content/anime ]; then
            cp -r _data/src/content/anime/. src/content/anime/
          fi
          echo "Anime count: $(ls src/content/anime/*.md 2>/dev/null | wc -l)"

      - name: Setup Node
        uses: actions/setup-node@v4
        with:
          node-version: 24
          cache: npm

      - name: Install
        run: npm install

      - name: Build
        run: npm run build

      - name: Deploy
        uses: cloudflare/wrangler-action@v4
        with:
          apiToken: ${{ secrets.CF_API_TOKEN }}
          accountId: ${{ secrets.CF_ACCOUNT_ID }}
          command: pages deploy dist --project-name=api-qimochi --branch=${{ github.ref_name }} --commit-dirty=true

name: Dump repo

on:
  workflow_dispatch:
    inputs:
      max_kb:
        description: Batas ukuran per file (KB)
        default: '200'

permissions:
  contents: write

jobs:
  dump:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - name: Gabungkan isi repo
        env:
          MAX_KB: ${{ inputs.max_kb }}
        run: |
          OUT=dump.txt
          MAX=$(( ${MAX_KB:-200} * 1024 ))
          SKIP='(^|/)(package-lock\.json|pnpm-lock\.yaml|yarn\.lock|bun\.lockb?)$|^src/content/anime/|^src/data/anime/|^src/data/actors/|^src/data/genres\.json$|^src/data/studios\.json$'

          {
            echo "# Repo: $GITHUB_REPOSITORY"
            echo "# Commit: $GITHUB_SHA"
            echo
            echo "## Struktur file"
            git ls-files | grep -Ev "$SKIP"
            echo
            echo "## Isi file"
          } > "$OUT"

          : > skipped.txt

          git ls-files -z | while IFS= read -r -d '' f; do
            if echo "$f" | grep -Eq "$SKIP"; then continue; fi
            [ -f "$f" ] || continue

            if ! grep -Iq . "$f"; then
              echo "biner atau kosong: $f" >> skipped.txt
              continue
            fi

            size=$(wc -c < "$f")
            if [ "$size" -gt "$MAX" ]; then
              echo "terlalu besar ($size byte): $f" >> skipped.txt
              continue
            fi

            {
              echo
              echo "===== FILE: $f ====="
              cat "$f"
            } >> "$OUT"
          done

          {
            echo
            echo "## File yang dilewati"
            cat skipped.txt
          } >> "$OUT"

          {
            echo "### Dump selesai"
            echo "Ukuran: $(wc -c < "$OUT") byte"
            echo "Dilewati: $(wc -l < skipped.txt) file"
          } >> "$GITHUB_STEP_SUMMARY"

      - name: Simpan hasil ke branch dump
        run: |
          git config user.name "github-actions[bot]"
          git config user.email "41898282+github-actions[bot]@users.noreply.github.com"

          git checkout --orphan dump
          git rm -rf --quiet .
          git add dump.txt
          git commit -m "Dump repo @ ${GITHUB_SHA::7}"
          git push --force origin dump

          echo "File: https://github.com/$GITHUB_REPOSITORY/blob/dump/dump.txt" >> "$GITHUB_STEP_SUMMARY"

import { defineCollection } from 'astro:content';
import { z } from 'astro/zod';
import { glob } from 'astro/loaders';

const Slug = z.string().regex(/^[a-z0-9]+(?:-[a-z0-9]+)*$/);
const ExternalUrl = z.url();

const AnimeType = z.enum(['TV', 'Movie', 'OVA', 'ONA', 'Special', 'Music', 'Unknown']);
const AnimeStatus = z.enum(['airing', 'finished', 'upcoming', 'hiatus', 'cancelled']);
const AnimeSource = z.enum([
  'original', 'manga', 'light_novel', 'visual_novel', 'game',
  'web_manga', 'web_novel', 'novel', 'book', 'picture_book',
  'radio', 'music', '4_koma_manga', 'card_game', 'other',
]);
const AnimeSeason = z.enum(['winter', 'spring', 'summer', 'fall']);
const AnimeRating = z.enum(['G', 'PG', 'PG-13', 'R', 'R+', 'Rx']);

const anime = defineCollection({
  loader: glob({
    pattern: ['**/*.md', '!**/_*.md'],
    base: './src/content/anime',
    deferRender: true,
  }),
  schema: z.object({
    title: z.string().min(1),
    titleEnglish: z.string().optional(),
    titleNative: z.string().optional(),
    malId: z.number().int().positive().optional(),
    anilistId: z.number().int().positive().optional(),
    kitsuId: z.string().optional(),
    type: AnimeType,
    status: AnimeStatus,
    source: AnimeSource.optional(),
    season: AnimeSeason.optional(),
    year: z.number().int().min(1900).max(2100).optional(),
    episodes: z.number().int().positive().nullable().optional(),
    duration: z.number().int().positive().optional(),
    rating: AnimeRating.optional(),
    aired: z.object({
      from: z.coerce.date().optional(),
      to: z.coerce.date().nullable().optional(),
    }).optional(),
    stats: z.object({
      score: z.number().min(0).max(10).optional(),
      scoredBy: z.number().int().nonnegative().optional(),
    }).optional(),
    genres: z.array(Slug).default([]),
    studios: z.array(Slug).default([]),
    image: ExternalUrl.optional(),
    banner: ExternalUrl.optional(),
    trailer: z.string().optional(),
    draft: z.boolean().default(false),
    addedAt: z.coerce.date().optional(),
    updatedAt: z.coerce.date().optional(),
  }),
});

export const collections = { anime };

===== FILE: src/lib/anime.ts =====
import { getCollection, type CollectionEntry } from 'astro:content';

export type AnimeEntry = CollectionEntry<'anime'>;

export async function getAllAnime(): Promise<AnimeEntry[]> {
  return getCollection('anime', ({ data }) => !data.draft);
}

export async function getAnimeById(id: string): Promise<AnimeEntry | null> {
  const all = await getAllAnime();
  return all.find((a) => a.id === id) ?? null;
}

export function animeToApi(entry: AnimeEntry) {
  const d = entry.data;
  return {
    id: entry.id,
    title: d.title,
    titleEnglish: d.titleEnglish ?? null,
    titleNative: d.titleNative ?? null,
    image: d.image ?? null,
    banner: d.banner ?? null,
    type: d.type,
    status: d.status,
    source: d.source ?? null,
    season: d.season ?? null,
    year: d.year ?? d.aired?.from?.getFullYear() ?? null,
    episodes: d.episodes ?? null,
    duration: d.duration ?? null,
    rating: d.rating ?? null,
    genres: d.genres ?? [],
    studios: d.studios ?? [],
    score: d.stats?.score ?? null,
    trailer: d.trailer ?? null,
  };
}

===== FILE: src/pages/api/v1/anime.json.ts =====
import type { APIRoute } from 'astro';
import { getAllAnime, animeToApi } from '~/lib/anime';

export const prerender = true;

export const GET: APIRoute = async () => {
  const all = await getAllAnime();

  const data = all
    .map(animeToApi)
    .sort((a, b) => {
      const ta = a.year ?? 0;
      const tb = b.year ?? 0;
      if (ta !== tb) return tb - ta;
      return a.title.localeCompare(b.title);
    });

  return new Response(
    JSON.stringify(
      {
        data,
        total: data.length,
        generated_at: new Date().toISOString(),
      },
      null,
      2
    ),
    {
      headers: {
        'Content-Type': 'application/json; charset=utf-8',
        'Cache-Control': 'public, max-age=3600, s-maxage=86400, stale-while-revalidate=604800',
        'Access-Control-Allow-Origin': '*',
      },
    }
  );
};

import type { APIRoute } from 'astro';
import { getAllAnime, animeToApi } from '~/lib/anime';

export const prerender = true;

export async function getStaticPaths() {
  const all = await getAllAnime();
  return all.map((entry) => ({
    params: { slug: entry.id },
    props: { entry },
  }));
}

export const GET: APIRoute = async ({ props }) => {
  const { entry } = props as { entry: any };
  const data = animeToApi(entry);

  return new Response(
    JSON.stringify({ data }, null, 2),
    {
      headers: {
        'Content-Type': 'application/json; charset=utf-8',
        'Cache-Control': 'public, max-age=3600, s-maxage=86400, stale-while-revalidate=604800',
        'Access-Control-Allow-Origin': '*',
      },
    }
  );
};
